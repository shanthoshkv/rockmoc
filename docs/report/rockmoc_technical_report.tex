\begin{abstract}
\noindent rockmoc designs supersonic nozzle contours (minimum-length, ideal, truncated, Rao
thrust-optimised, conical, planar NASA, annular aerospike) by the method of characteristics
(MOC), and estimates their performance with an exit-plane field integral, a momentum-integral
boundary layer and a JANNAF-style loss stack. Thermochemistry ranges from perfect gas through
thermally perfect multi-$\gamma$ isentropes to NASA CEA and Cantera. This document is the single
reference for the project: theory, every public function, worked examples, all figures,
validation and limits.
\end{abstract}

\section{Overview}
\textbf{Purpose.} Education and preliminary design of rocket and test nozzles; generate
CAD/CFD-ready walls (CSV, JSON, DXF, NumPy, OpenFOAM \code{blockMeshDict}).
\textbf{Not} a CFD solver, flight-qualification tool or combustion model.

\textbf{Install.}
\begin{lstlisting}[language=bash]
pip install rockmoc                    # core: numpy, scipy
pip install "rockmoc[plot,fast,cea]"   # matplotlib, numba kernels, RocketCEA
pip install "rockmoc[all]"             # + plotly, cantera, coolprop, dev tools
\end{lstlisting}
Extras: \code{plot}, \code{fast} (Numba), \code{viz} (Plotly), \code{cea}, \code{cantera},
\code{coolprop}, \code{dev}, \code{all}. Python $\ge$ 3.10, MIT licence.

\textbf{Quick start.}
\begin{lstlisting}
from rockmoc import DesignSpec, PerfectGas, design_rao
spec = DesignSpec(gas=PerfectGas(1.2, 350.0), throat_radius=0.05,
                  exit_mach=3.0, n_characteristics=60)
r = design_rao(spec, length_fraction=0.8)
print(r.summary()["area_ratio"], r.metrics.length)
r.export_csv("rao.csv"); r.plot(save_dir="figs")
bl = r.with_boundary_layer(pc=7e6, Tc=3400, wall_temperature=800)
print(bl.meta["boundary_layer"]["delta_cf_friction"])
\end{lstlisting}

\textbf{CLI.}
\begin{lstlisting}[language=bash]

\section{Theory and numerics}
\subsection{Assumptions}
Steady, inviscid, irrotational, isentropic core; planar ($\delta=0$) or axisymmetric
($\delta=1$); shock-free design region; frozen or prescribed-$\gamma$ composition; viscous
effects added afterwards (BL, \S\ref{sec:bl}).

\subsection{Thermodynamics}
Perfect gas (\code{PerfectGas(gamma, R)}):
\[
\frac{T_0}{T}=1+\tfrac{\gamma-1}{2}M^2,\quad
\frac{A}{A^*}=\frac1M\Big[\tfrac{2}{\gamma+1}\big(1+\tfrac{\gamma-1}{2}M^2\big)\Big]^{\frac{\gamma+1}{2(\gamma-1)}},\quad
\nu=\sqrt{\tfrac{\gamma+1}{\gamma-1}}\tan^{-1}\!\sqrt{\tfrac{\gamma-1}{\gamma+1}(M^2-1)}-\tan^{-1}\!\sqrt{M^2-1}.
\]
$c^*=\sqrt{RT_0/\gamma}\,\big(\tfrac{\gamma+1}{2}\big)^{\frac{\gamma+1}{2(\gamma-1)}}$,
$C_F=\sqrt{\tfrac{2\gamma^2}{\gamma-1}\big(\tfrac{2}{\gamma+1}\big)^{\frac{\gamma+1}{\gamma-1}}\big(1-(p_e/p_c)^{\frac{\gamma-1}{\gamma}}\big)}+\varepsilon\frac{p_e-p_a}{p_c}$.
Inverses $M(A/A^*)$, $M(\nu)$ use safeguarded Newton with analytic derivatives (Numba kernels
when installed).

\textbf{Multi-$\gamma$ (thermally perfect) isentrope} (\code{build_multi_gamma_table}):
$\gamma(M)$ linear $\gamma_c\to\gamma_t$ on $M\in[0,1]$ and $\gamma_t\to\gamma_e$ on $[1,M_{max}]$;
integrated from stagnation with $c_p=\gamma R/(\gamma-1)$:
\[
dh=c_p\,dT,\quad d\ln p=\frac{c_p\,dT}{RT},\quad V^2=2(h_c-h),\quad
\frac{A}{A^*}=\frac{(\rho V)^*}{\rho V},\quad d\nu=\sqrt{M^2-1}\,\frac{dV}{V}.
\]
$\rho V$ peaks at $M=1$ for any such gas. Constant $\gamma$ reproduces perfect gas to $10^{-7}$.
\code{VariableGammaGas} uses the table's own $c^*=p_c/(\rho V)^*$ and the exact
$C_F=\tfrac{A_e}{A^*}\tfrac{p_e}{p_c}\gamma_eM_e^2+\varepsilon\tfrac{p_e-p_a}{p_c}$ (from $\rho V^2=\gamma pM^2$).

\textbf{Chemistry backends}: \code{RocketCEAGas} (CEA, freeze at throat/chamber),
\code{CanteraGas} (HP equilibrium, frozen $\gamma$), \code{CoolPropGas},
\code{build_cea_frozen_table}. All satisfy the \code{MOCGas} protocol.

\subsection{Method of characteristics}
Characteristics $dr/dx=\tan(\theta\pm\mu)$, $\mu=\sin^{-1}(1/M)$; compatibility
\[
d(\theta\mp\nu)=\pm\,\delta\,\frac{\sin\theta\sin\mu}{r\cos(\theta\pm\mu)}\,dx\quad\text{along }C^\pm .
\]
Unit processes (\code{moc.mesh}): interior, axis ($\theta=0$, reached along $C^-$), wall
(prescribed angle or streamline). Second-order predictor--corrector on averaged
$\theta,\mu,r$. Walls are streamlines carrying the throat mass flow; transition regions are
Goursat problems to the uniform exit $C^+$.

\subsection{Transonic initial line}
\textbf{Hall (1962), 2nd order} --- default for axisymmetric throats. With $R=R_c/R_t$,
$\varepsilon=1/R$, lengths$/R_t$, $z=x\sqrt{2R/(\gamma+1)}$:
\[
u=1+\varepsilon u_1+\varepsilon^2u_2,\qquad v=\sqrt{\tfrac{(\gamma+1)\varepsilon}{2}}\,(\varepsilon v_1+\varepsilon^2v_2),
\]
\[
u_1=\tfrac{r^2}{2}-\tfrac14+z,\quad v_1=\tfrac{r^3}{4}-\tfrac r4+rz,\quad
u_2=\tfrac{2\gamma+9}{24}r^4-\tfrac{4\gamma+15}{24}r^2+\tfrac{10\gamma+57}{288}+z(r^2-\tfrac58)-\tfrac{2\gamma-3}{6}z^2,
\]
\[
v_2=\tfrac{\gamma+3}{9}r^5-\tfrac{20\gamma+63}{96}r^3+\tfrac{28\gamma+93}{288}r+z\big(\tfrac{2\gamma+9}{6}r^3-\tfrac{4\gamma+15}{12}r\big)+rz^2 .
\]
Residuals of the exact potential equation fall as $\varepsilon^{3.4}$ (Sauer:
$\varepsilon^{2.5}$), but the series diverges for tight throats (throat-plane axis $M$ 1.30
at $R_c/R_t=0.625$ vs 0.8 measured \citep{cuffel1969}). Hall is used for $R_c/R_t\ge1.5$
(wall $p/p_0$ at $R_c/R_t=2$ within 0.3\% of NASA TN D-5377 vs 5\% for Sauer), Sauer below.
\textbf{Sauer} (first order, planar default):
$\alpha=\sqrt{(1+\delta)/((\gamma+1)R_cR_t)}$, $u=\alpha x+\frac{(\gamma+1)\alpha^2r^2}{2(1+\delta)}$.
Fallback chain: Hall $\to$ Sauer $\to$ uniform near-sonic line (warned).

\subsection{Contour families}
\begin{description}
\item[MLN] sharp-corner minimum-length nozzle; $\theta_{max}$ solved so the last corner $C^-$
  hits the axis at $M_e$ (axisymmetric $\theta_{max}<\nu_e/2$; planar $=\nu_e/2$).
\item[Ideal] circular-arc throat + characteristic lattice to uniform, parallel exit.
\item[TIC] ideal contour truncated to a length fraction.
\item[Rao TOC] (\code{design_rao_control}, core \code{moc.rao.design_rao_moc}) control surface $DE$ with
  $V\cos(\theta-\mu)/\cos\mu=\lambda$, $r^\delta\rho V^2\sin^2\theta\tan\mu=K$, mass balance
  $\dot m_{DE}=\dot m_{DN}$; $D$ chosen for $r_E=r_e$, $\theta_N$ for $L=fL_{15}$.
  \emph{v0.8 fix:} $(\theta,M)$ on $DE$ solved by a $2\times2$ Newton; the old
  $\theta=\mu\pm\cos^{-1}(\cdot)$ elimination was singular at $\theta=\mu$, which the optimum
  crosses near $f\approx0.8$. $\theta_N$ now converges monotonically
  (27.47$\to$26.95$^\circ$, $n=20\to120$, $\gamma=1.2$, $\varepsilon=13.8$).
\item[Rao TOP] \code{design_rao} (parabola with chart $\theta_N,\theta_e$) and
  \code{design_rao_toc} (same parabola, length scaled to the MOC ideal length; \emph{not} a
  control-surface solution).
\item[Conical] 15$^\circ$ (or user) cone with throat arc; $\eta_{div}=(1+\cos\alpha)/2$.
\item[Planar NASA] \code{design_planar_mln_nasa}: TM-X-1502 sharp-throat 2-D MLN.
\item[Aerospike] Lee / Vernacchia spike (\code{design_spike}, truncated, base pressure, cowl).
\end{description}
Length ordering for equal $M_e$: Rao $<$ TIC $<$ MLN $<$ ideal (\S\ref{sec:figs}, family overlay).

\subsection{Boundary layer (v0.8)}\label{sec:bl}
\code{correct_wall_bl(method="integral")} solves the compressible von K\'arm\'an equation
along wall arc length $s$ with an isentropic edge state:
\[
\frac{d\theta}{ds}=\frac{C_f}{2}-\theta\Big[\frac{2+H-M_e^2}{1+\frac{\gamma-1}{2}M_e^2}\,\frac{1}{M_e}\frac{dM_e}{ds}+\frac{\delta}{r}\frac{dr}{ds}\Big].
\]
Closures: reference temperature $T^*/T_e=1+0.032M^2+0.58(T_w/T_e-1)$;
$C_f=(\rho^*/\rho_e)\,C_{f,inc}(Re_\theta^*)$ with $C_{f,inc}=0.0256Re_\theta^{-1/4}$ (turbulent)
or $0.441/Re_\theta$ (laminar); Crocco--Busemann shape factor $H=H_iT_w/T_e+r\frac{\gamma-1}{2}M^2$
($H_i=9/7$, 2.59); recovery $r=Pr^{1/3}$; Bartz transport $\mu=1.184\times10^{-7}\mathcal M^{0.5}T^{0.6}$,
$Pr=4\gamma/(9\gamma-5)$. Adiabatic wall unless \code{wall_temperature} given. Integrated by
LSODA; on failure it falls back (with \code{MOCFallbackWarning}) to the zero-pressure-gradient
closed form of the same closure, $\theta=[(n+1)\tfrac a2\tfrac{\rho^*}{\rho_e}(\tfrac{\mu^*}{\rho^*U})^ns]^{1/(n+1)}$,
which equals the ODE result at $dM/ds=dr/ds=0$. Wall offset $\delta^*=H\theta$ along the normal
(TM-X-2343 rule). Friction loss $\Delta C_F=\int\tau_w\cos\alpha\,2\pi r\,ds/(p_cA_t)$.
$C_f$ agrees with Van Driest~II within 3\% for $M\le5$, adiabatic and $T_w/T_{aw}=0.5$.

\subsection{Performance and loss stack}
1-D metrics (\code{compute_metrics}); exit-plane field integral (\code{field_integrated_cf}):
$F=\int(\rho u^2+p)2\pi r\,dr-p_aA_e$, $C_F=F/(\dot mc^*)$ on the MOC exit ray.
Integrated estimate:
\[
C_F=C_{F,1D}\,\eta_{div}\,\eta_{ke}\,\eta_{chem}\,\eta_{2\phi}-\Delta C_{F,friction}.
\]
$\eta_{chem}$ bounds from \code{kinetic_efficiency_bounds}: frozen-at-throat/shifting Isp
(LOX/RP-1, $O/F=2.6$, 1000~psia, $\varepsilon=40$: 0.945--1.0). Altitude via
\code{pressure_at_altitude} (exponential/US-76-style to 80~km).

\section{Examples}
\textbf{Family comparison.}
\begin{lstlisting}
from rockmoc import DesignSpec, PerfectGas, compare_nozzle_designs
cmp = compare_nozzle_designs(DesignSpec(gas=PerfectGas(1.2, 350), exit_mach=3, n_characteristics=40))
print(cmp.summary_table())          # L/Rt, eps, Cf_vac, eta_n per family
\end{lstlisting}
\textbf{Integral BL with fallback, cooled wall, loss stack.}
\begin{lstlisting}
from rockmoc import correct_wall_bl, design_mln, integrated_performance
r  = design_mln(spec)
bl = correct_wall_bl(r.contour, spec.gas, throat_radius=0.05, pc=7e6, Tc=3400,
                     method="integral", wall_temperature=800.0)
print(bl.meta["method"], bl.delta_star[-1], bl.delta_cf_friction)
perf = integrated_performance(bl.corrected, spec.gas, throat_radius=0.05, exit_mach=3,
                              pc=7e6, Tc=3400, delta_cf_friction=bl.delta_cf_friction,
                              eta_chemical=0.97)
print(perf.meta["loss_stack"])
\end{lstlisting}

\begin{lstlisting}

\section{Limits addressed in v0.9}
Each former limit now has a model and a validation gate (\code{rockmoc.validate.experimental}, 74/74):
\begin{description}
\item[Tight throats] Kliegel--Levine series $u=1+\varepsilon u_1+\varepsilon^2(u_1+u_2)+\varepsilon^3(u_1+2u_2+u_3)$,
  $\varepsilon=1/(1+R)$, for $0.4\le R<1.5$ (Hall's $z=x\sqrt{2R/(\gamma+1)}$ is the only scaling that keeps
  $\partial u/\partial r=\partial v/\partial x$ order by order); closed-form $C_d$. Below 0.4 a 2-D Euler
  solver (\code{moc.transonic_euler}, first-order Rusanov FV, Richardson-extrapolated). Cuffel $R=0.625$:
  KL axis $M$ 0.79, $C_d$ 0.982; Euler 0.82, 0.982; measured 0.8, 0.985.
\item[Boundary layer] Bartz energy integral (interaction exponent 0)
  $d\phi/ds=St\,(T_{aw}-T_w)/(T_0-T_w)-\phi[d\ln\rho U/ds+d\ln r/ds+d\ln(T_0-T_w)/ds]$, two-layer
  $1/7$ profiles over $\delta$ and $\Delta$ giving $H(M,T_w/T_0,\Delta/\delta)$ (can be negative), chamber-inlet
  start, and the relaminarisation switch $K_{ax}>2.88\times10^{-6}$ \citep{boldman1969}.
\item[Separation] criteria from Summerfield to Stark, truncated-nozzle or Kalt--Badal recovery $C_F$
  (identical to RocketCEA), NASA TP-1207 guards.
\item[Internal shocks] direct MOC of a given wall with a cell-Jacobian crossing test: TOP parabolas
  and arc-throat cones flagged, ideal contours clean.
\item[Kinetics] quasi-1-D finite-rate Cantera nozzle; frozen and equilibrium limits reproduced exactly;
  $\eta_{kin}$ 0.980/0.991/0.997 for 5\,mm/5\,cm/50\,cm H$_2$/O$_2$ throats.
\item[Two-phase] gas--particle velocity/thermal lag; no-lag limit equals the mixture isentrope.
\item[MLN data] Zebbiche's perfect-gas axisymmetric MLN table: $L/r^*$ within 0.03\%, angles within 0.02$^\circ$.
\end{description}

\section{Limits and recommended use}
\begin{itemize}
\item Inviscid core outside the throat; separation and internal shocks are detected/estimated, not
  resolved (no shock fitting, no restricted-shock-separation pattern).
\item BL: integral method; cooled-throat $\delta^*\approx0$ is sign-sensitive; no film cooling.
\item KL start line is vertical; at $R<0.6$ its wall Mach rises steeply downstream, use the Euler solver there.
\item Kinetics: reaction mechanism supplied by the user (H$_2$/O$_2$ default); equilibrium start at
  $\varepsilon=1.05$. Two-phase: particle size is an input; starts in equilibrium at $\varepsilon=1.05$ (slightly optimistic).
\item Separation criteria scatter by $\sim$20\%; report Schmucker and Stark as a band.
\end{itemize}
Workflow: choose family $\to$ \code{design_*} at $n\approx40$ $\to$ check convergence
$\to$ \code{design_production} with CEA $\gamma$, wall $T$, $\eta_{chem}$ $\to$ export DXF/OpenFOAM
$\to$ CFD verification.
